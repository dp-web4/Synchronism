% Glyphs Latin Modern (the default xelatex font) does not carry. Without this
% xelatex drops them SILENTLY: measured 2026-10-07, 3,725 missing-character
% warnings over 75 code points: gamma, rho, sigma, the sub/superscripts, nabla,
% approx and the rest of the physics. "freeing gamma from MOND's 1/2" printed
% as "freeing   from MOND's 1/2".
%
% Each listed character is drawn from FreeSerif in text and FreeMono in code
% (fonts-freefont-otf), by code point (\symbol), so the active character does
% not re-expand. Latin Modern stays the face for everything it has. The
% mappings are deferred to \begin{document} because unicode-math redefines
% some of these characters there.
%
% make-pdf.sh prints any remaining "Missing character" warning. Add a line here
% for each new one: \wpfallback in general, \wpfallbackserif if FreeMono (any weight) lacks it.
\usepackage{newunicodechar}
\newfontfamily\wpfbserif{FreeSerif}
\newfontfamily\wpfbmono{FreeMono}
\makeatletter
\AtBeginDocument{\begingroup\ttfamily\xdef\wp@ttfamily{\f@family}\endgroup}
\newcommand\wpfallback[1]{{\ifx\f@family\wp@ttfamily\wpfbmono\else\wpfbserif\fi\symbol{#1}}}
\makeatother
\newcommand\wpfallbackserif[1]{{\wpfbserif\symbol{#1}}}
